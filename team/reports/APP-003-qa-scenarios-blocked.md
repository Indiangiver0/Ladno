```powershell
node -e "const fs=require('node:fs'),c=require('node:crypto'),s=JSON.parse(fs.readFileSync('team/snapshots/local-prototype-v1.json','utf8')); const a=s.files.slice().sort((a,b)=>a.path<b.path?-1:a.path>b.path?1:0); const bad=a.filter(e=>c.createHash('sha256').update(fs.readFileSync(e.path)).digest('hex')!==e.sha256); const digest=c.createHash('sha256').update(a.map(e=>e.sha256+'  '+e.path+'\n').join('')).digest('hex'); console.log(JSON.stringify({version:s.version,fileCount:a.length,mismatches:bad.map(e=>e.path),digest},null,2)); if(bad.length||digest!=='be7ef8f3e912744b04804a7a225fdee44ad6917254de68529f97a2af728e354e')process.exitCode=1;"
```
